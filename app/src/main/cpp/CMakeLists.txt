# Fase 2B: proot propio rename-safe, modo loader bundled, solo arm64-v8a.
#
# Espejo del flujo bundled de proot/src/GNUmakefile (termux/proot):
#   1. loader estatico (loader.c + assembly.S, -ffreestanding -static
#      -nostdlib, -Ttext=LOADER_ADDRESS arm64 0x2000000000)
#   2. llvm-strip -> loader.exe
#   3. llvm-objcopy -I binary -> loader-wrapped.o (embebido en proot)
#   4. llvm-readelf -s loader | gen_loader_info.py -> loader-info.c
#      (sustituye loader-info.awk: necesita gawk/strtonum y solo hay mawk)
#   5. link final proot = objetos proot + talloc + loader-wrapped.o
#      + loader-info.o, SIN -DPROOT_UNBUNDLE_LOADER (rename-safe).
#
# Runtime: lanzar con PROOT_TMP_DIR=<filesDir> (el loader embebido se
# extrae a un temporal y /tmp es noexec en Android).

cmake_minimum_required(VERSION 3.22.1)

project("proot-bundled" C)

# Solo arm64-v8a (Gradle abiFilters lo impone; el loader embebido lleva
# LOADER_ADDRESS y formato/objcopy fijos de AArch64).
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "proot bundled build supports only arm64-v8a (got ${ANDROID_ABI})")
endif()

find_package(Python3 REQUIRED COMPONENTS Interpreter)

set(PROOT_SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/proot)
set(TALLOC_SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/talloc)
set(GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${GEN_DIR})

# Herramientas llvm del NDK (el toolchain Android no siempre expone
# CMAKE_OBJCOPY/CMAKE_STRIP/CMAKE_READELF: derivar del compilador).
get_filename_component(NDK_BIN_DIR "${CMAKE_C_COMPILER}" DIRECTORY)
set(LLVM_OBJCOPY ${NDK_BIN_DIR}/llvm-objcopy)
set(LLVM_STRIP ${NDK_BIN_DIR}/llvm-strip)
set(LLVM_READELF ${NDK_BIN_DIR}/llvm-readelf)

# build.h: features verificados en Fase 2A (arm64 Android 28, NDK r30).
file(WRITE ${GEN_DIR}/build.h
    "/* Auto-generated by Fase 2B CMake (fixed features, arm64 Android). */\n"
    "#ifndef BUILD_H\n"
    "#define BUILD_H\n"
    "#define VERSION \"5.1.107\"\n"
    "#define HAVE_PROCESS_VM\n"
    "#define HAVE_SECCOMP_FILTER\n"
    "#endif /* BUILD_H */\n"
)

set(TALLOC_SOURCES
    talloc/talloc.c
)

# Paso 1: objetos del loader con flags freestanding.
add_custom_command(
    OUTPUT ${GEN_DIR}/loader.o
    COMMAND ${CMAKE_C_COMPILER} -D_FILE_OFFSET_BITS=64 -D_GNU_SOURCE
        -I${PROOT_SRC_DIR} -c ${PROOT_SRC_DIR}/loader/loader.c
        -o ${GEN_DIR}/loader.o -fPIC -ffreestanding
    DEPENDS ${PROOT_SRC_DIR}/loader/loader.c
    COMMENT "Compiling proot loader.c"
)
add_custom_command(
    OUTPUT ${GEN_DIR}/assembly.o
    COMMAND ${CMAKE_C_COMPILER} -D_FILE_OFFSET_BITS=64 -D_GNU_SOURCE
        -I${PROOT_SRC_DIR} -c ${PROOT_SRC_DIR}/loader/assembly.S
        -o ${GEN_DIR}/assembly.o -fPIC -ffreestanding
    DEPENDS ${PROOT_SRC_DIR}/loader/assembly.S
    COMMENT "Compiling proot assembly.S"
)

# Paso 1b: link estatico del loader (LOADER_ADDRESS arm64 = 0x2000000000).
add_custom_command(
    OUTPUT ${GEN_DIR}/loader
    COMMAND ${CMAKE_C_COMPILER} -o ${GEN_DIR}/loader
        ${GEN_DIR}/loader.o ${GEN_DIR}/assembly.o
        -static -nostdlib
        -Wl,--build-id=none,-Ttext=0x2000000000,--rosegment,-z,noexecstack
    DEPENDS ${GEN_DIR}/loader.o ${GEN_DIR}/assembly.o
    COMMENT "Linking proot loader (static, Ttext=0x2000000000)"
)

# Paso 2: strip -> loader.exe (equivale al cp+strip del GNUmakefile).
add_custom_command(
    OUTPUT ${GEN_DIR}/loader.exe
    COMMAND ${LLVM_STRIP} ${GEN_DIR}/loader -o ${GEN_DIR}/loader.exe
    DEPENDS ${GEN_DIR}/loader
    COMMENT "Stripping proot loader"
)

# Paso 3: objcopy binario -> loader-wrapped.o (bundled, sin path).
# OJO: objcopy deriva _binary_<fichero>_start del nombre tal cual se pasa;
# usar nombres relativos (como el GNUmakefile) para obtener
# _binary_loader_exe_start/_end/_size que espera execve/enter.c.
add_custom_command(
    OUTPUT ${GEN_DIR}/loader-wrapped.o
    COMMAND ${LLVM_OBJCOPY}
        --input-target=binary
        --output-target=elf64-littleaarch64
        --binary-architecture=aarch64
        loader.exe loader-wrapped.o
    DEPENDS ${GEN_DIR}/loader.exe
    WORKING_DIRECTORY ${GEN_DIR}
    COMMENT "Embedding proot loader (objcopy bundled)"
)

# Paso 4: loader-info.c via python (mawk no tiene strtonum).
add_custom_command(
    OUTPUT ${GEN_DIR}/loader-info.c
    COMMAND ${LLVM_READELF} -s ${GEN_DIR}/loader
        | ${Python3_EXECUTABLE} ${PROOT_SRC_DIR}/loader/gen_loader_info.py
        > ${GEN_DIR}/loader-info.c
    DEPENDS ${GEN_DIR}/loader ${PROOT_SRC_DIR}/loader/gen_loader_info.py
    COMMENT "Generating loader-info.c"
)

# Paso 5: objetos proot + talloc + loader-info generado.
add_library(proot_objects OBJECT
    ${PROOT_SOURCES}
    ${TALLOC_SOURCES}
    ${GEN_DIR}/loader-info.c
)
target_include_directories(proot_objects PRIVATE
    ${PROOT_SRC_DIR}
    ${TALLOC_SRC_DIR}
    ${GEN_DIR}
)
target_compile_definitions(proot_objects PRIVATE
    _FILE_OFFSET_BITS=64
    _GNU_SOURCE
    __STDC_WANT_LIB_EXT1__=1
)
target_compile_options(proot_objects PRIVATE
    -O2 -Wall -Wextra -Wno-unused-parameter -Wno-sign-compare
)

add_executable(proot
    $<TARGET_OBJECTS:proot_objects>
    ${GEN_DIR}/loader-wrapped.o
)
target_link_options(proot PRIVATE -Wl,-z,noexecstack)

# Empaquetado: el ejecutable no va a jniLibs; se publica como asset para
# que TermuxInstaller (Fase 3) lo copie a files/bin/proot con 0700.
add_custom_command(TARGET proot POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory
        ${CMAKE_CURRENT_SOURCE_DIR}/../assets/arm64-v8a
    COMMAND ${CMAKE_COMMAND} -E copy
        $<TARGET_FILE:proot>
        ${CMAKE_CURRENT_SOURCE_DIR}/../assets/arm64-v8a/proot
    COMMENT "Staging proot into src/main/assets/arm64-v8a/proot"
)
